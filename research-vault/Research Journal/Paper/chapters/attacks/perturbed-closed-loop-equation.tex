\begin{definition}
	Perturbed Closed-Loop Tracking Control State Equation
	\begin{equation}
		\begin{aligned}
			 & x(k+1) = \state - \beta \text{sgn}(x(k) - r(k)) + \Delta r(k) + I(k) \\
			 & \text{where:}                                                        \\
			 & I(k) \sim \mathcal N(0, \sigma^2 \mathcal Id_n)                      \\
		\end{aligned}
	\end{equation}

\end{definition}
