\subsection{Convergence Analysis}

\begin{theorem}[Defended Tracking Error Bound]
	\label{thm:defended_convergence}
	Consider the defended system \eqref{eq:defended_closed_loop} with $n = 3$
	processing units \eqref{eq:channel_error}, where exactly one unit is corrupted with
	noise $\text{std} > 0$ and the remaining two are ideal
	($\delta_i(k) = 0$). Then

	\begin{enumerate}
		\item \textbf{(Detection)} The minimum-variance pair selection
		      \eqref{eq:pair_selection} deterministically excludes the corrupted unit
		      at every time step.

		\item \textbf{(Exact recovery)} When the corrupted unit is excluded, the
		      defended error satisfies
		      \begin{equation}
			      \label{eq:defended_error_bound}
			      \hat{e}(k) = e(k)
		      \end{equation}
		      with zero residual noise.

		\item \textbf{(Tracking convergence)} The defended closed-loop
		      \eqref{eq:defended_closed_loop} preserves
		      $\operatorname{sgn}(\hat{e}_n(k)) = \operatorname{sgn}(e_n(k))$ for all
		      nonzero components and drives the tracking error toward zero, recovering
		      nominal performance.
	\end{enumerate}
\end{theorem}
