
\subsubsection*{\centering Convergence of Steady States}

\begin{theorem}\label{thm:convergence}
	Starting from any finite initial error $e(1) = e_1$ with $|e_{1,i}|<\infty$,
	the tracking error $e_i(k)$ governed by~\autoref{eq:error_dynamics} converges to
	$e_i^{*} = 0$ in at most
	\begin{equation}
		K_i = \left\lceil\frac{|e_{1,i}|}{\beta_i}\right\rceil
		\label{eq:convergence_bound}
	\end{equation}
	time steps.  The rate of convergence is directly tunable by
	$\beta_i$\,\textsubscript{\cite{tymoshchuk2024tracking}}.
\end{theorem}

\begin{proof}
	Consider the $i$-th component of~\autoref{eq:error_dynamics}:
	\begin{equation}
		e_i(k+1) = e_i(k) - \beta_i\,\text{sgn}(e_i(k)).
		\label{eq:scalar_error_dynamics}
	\end{equation}
	Let $V_i(k) = e_i(k)^{2}$ be a Lyapunov candidate.  For any $e_i(k)\neq 0$:
	\begin{align}
		V_i(k+1) - V_i(k)
		 & = \bigl[e_i(k) - \beta_i\,\text{sgn}(e_i(k))\bigr]^{2} - e_i(k)^{2}
		\notag                                                                 \\
		 & = -2\beta_i\,|e_i(k)| + \beta_i^{2}.
		\label{eq:lyapunov_delta}
	\end{align}
	When $|e_i(k)| \geq \beta_i$, equation~\autoref{eq:lyapunov_delta} gives
	$V_i(k+1)-V_i(k) \leq -\beta_i^{2} < 0$, so $V_i$ decreases strictly.
	Because $|e_i(k)|$ is reduced by $\beta_i$ at each such step, $|e_i(k)|$
	reaches zero in at most $K_i = \lceil|e_{1,i}|/\beta_i\rceil$ steps.
	A larger $\beta_i$ reduces $K_i$ and accelerates convergence; the convergence
	time can therefore be adjusted by varying $\beta_i$~\textsubscript{\cite{tymoshchuk2024tracking}}.
\end{proof}
